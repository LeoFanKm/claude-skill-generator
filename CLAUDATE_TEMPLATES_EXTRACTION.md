## 🎯 Usage in SkillGenerator

**Example integration in main generator**:

```typescript
import Mustache from 'mustache';
import { TEMPLATES } from './templates';

export class SkillGenerator {
  private renderSkillMd(request: SkillCreateRequest): string {
    const data = {
      name: request.name,
      description: request.description,
      displayName: request.displayName,
      overview: this.generateOverview(request),
      capabilities: request.capabilities.map((cap, i) => ({
        title: `Capability ${i + 1}`,
        description: cap
      })),
      inputRequirements: request.inputRequirements,
      outputFormats: request.outputFormats,
      usageExamples: this.generateUsageExamples(request),
      hasPythonFiles: request.includePython,
      pythonFiles: request.includePython ? [
        {
          filename: `${this.toSnakeCase(request.name)}.py`,
          description: `Main implementation for ${request.displayName}`
        }
      ] : [],
      bestPractices: this.generateBestPractices(request),
      limitations: this.generateLimitations(request)
    };

    return Mustache.render(TEMPLATES.skillMd, data);
  }

  private renderReadme(request: SkillCreateRequest): string {
    const data = {
      displayName: request.displayName,
      shortDescription: request.description,
      skillName: request.name,
      quickStartExample: `"Hey Claude, ${request.useCases[0]?.toLowerCase() || 'help me with my task'}"`,
      requirements: [
        'Claude Code CLI or Claude apps',
        request.includePython ? 'Python 3.8+' : null
      ].filter(Boolean),
      features: request.capabilities,
      examples: this.generateExamples(request),
      version: '1.0.0',
      lastUpdated: new Date().toISOString().split('T')[0],
      created: new Date().toISOString().split('T')[0]
    };

    return Mustache.render(TEMPLATES.readme, data);
  }

  // ... other methods
}
```
